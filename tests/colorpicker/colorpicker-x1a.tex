% Color Picker の出力インテント確認用サンプル (pdflatex を 2 回)。
% pdfx の x-1a で /OutputIntents (FOGRA39L Coated, colorprofiles 同梱) を書かせる。
% RGB や透明を含むので適合 PDF/X-1a ではない。欲しいのは /OutputIntents だけ。
%
% xcolor は pdfx より先に読む。pdfx は x-* で xcolor が未読込だと [cmyk] 付きで
% 読み込み、RGB の色指定を TeX 側で CMYK に変えてしまうため。
\documentclass[a4paper,11pt]{article}
\usepackage[margin=25mm]{geometry}
\usepackage[dvipsnames]{xcolor}
\usepackage{spotxcolor}
\usepackage[x-1a]{pdfx}
% pdfx は x-1a で xcolor の目標モデルを cmyk に固定し、\selectcolormodel も
% 警告だけの定義に差し替える。RGB の色を RGB のまま書かせたいので、xcolor の
% 内部状態を natural に戻す (xcolor の \selectcolormodel{natural} と同じ処理)。
\makeatletter
\def\XC@tgt@mod#1{#1}\convertcolorsDfalse\convertcolorsUfalse
\makeatother
\usepackage{tikz}
\usepackage{graphicx}
\pagestyle{empty}

\definecolor{cmykred}{cmyk}{0,1,1,0}
\definecolor{rgbblue}{rgb}{0,0.4,1}
\definecolor{gray50}{gray}{0.5}
\definecolor{richblack}{cmyk}{0.6,0.4,0.4,1}
\definespotcolor{DIC161s}{DIC 161s*}{0, 0.64, 1, 0}

\begin{document}

\noindent
Plain body text (DeviceGray 0).\quad
\textcolor{cmykred}{CMYK red text.}\quad
\textcolor{rgbblue}{RGB blue text.}\quad
\textcolor{gray50}{Gray 50\% text.}

\medskip\noindent
\textcolor{DIC161s}{Spot DIC 161s* 100\%.}\quad
\textcolor{DIC161s!50}{Spot DIC 161s* 50\%.}\quad
\textcolor{richblack}{Rich black text.}\quad
\textcolor[rgb]{0,0,0}{RGB black text.}

\begin{tikzpicture}[remember picture, overlay, x=1pt, y=1pt, shift={(current page.south west)}]
  % --- fills: 100x60 boxes, bottom-left at (x, 600) ---
  \fill[black]                  (60,600)  rectangle (160,660);   % DeviceGray 0
  \fill[gray50]                 (180,600) rectangle (280,660);
  \fill[rgbblue]                (300,600) rectangle (400,660);
  \fill[color={rgb,1:red,0;green,0;blue,0}] (420,600) rectangle (520,660); % DeviceRGB black

  % --- second row ---
  \fill[cmykred]                (60,520)  rectangle (160,580);
  \fill[richblack]              (180,520) rectangle (280,580);
  \fill[DIC161s]                (300,520) rectangle (400,580);
  \fill[DIC161s!50]             (420,520) rectangle (520,580);

  % --- images: 100pt wide, bottom-left at (x, 400) ---
  \node[anchor=south west, inner sep=0] at (60,400)  {\includegraphics[width=100pt]{photo-rgb.jpg}};
  \node[anchor=south west, inner sep=0] at (180,400) {\includegraphics[width=100pt]{photo-cmyk.jpg}};
\end{tikzpicture}

\end{document}
